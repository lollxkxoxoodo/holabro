    end)
end)

local function CreateActionBtn(text, pos, size, callback)
    local btn = Instance.new("TextButton")
    btn.Size = size
    btn.Position = pos
    btn.BackgroundColor3 = Color3.fromRGB(45, 45, 45)
    btn.Text = text
    btn.TextColor3 = Color3.fromRGB(255, 255, 255)
    btn.Font = Enum.Font.Gotham
    btn.TextSize = 13
    btn.AutoButtonColor = true
    btn.Parent = MainFrame
    
    local corner = Instance.new("UICorner")
    corner.CornerRadius = UDim.new(0, 6)
    corner.Parent = btn
    
    btn.MouseButton1Click:Connect(callback)
    return btn
end

local function GetClosestPlayerToCursor()
    local closestPlayer = nil
    local shortestDistance = math.huge
    local center = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)

    for _, player in pairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character and player.Character:FindFirstChild("Head") then
            local humanoid = player.Character:FindFirstChildOfClass("Humanoid")
            if humanoid and humanoid.Health > 0 and humanoid:GetState() ~= Enum.HumanoidStateType.Dead then
                local screenPos, onScreen = Camera:WorldToViewportPoint(player.Character.Head.Position)
                if onScreen then
                    local distance = (Vector2.new(screenPos.X, screenPos.Y) - center).Magnitude
                    if distance < shortestDistance and distance < Configuration.MaxLockDistance then
                        shortestDistance = distance
                        closestPlayer = player
                    end
                end
            end
        end
    end
    return closestPlayer
end

local function UpdateEsp()
    for _, player in pairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character then
            local head = player.Character:FindFirstChild("Head")
            if head then
                local highlight = player.Character:FindFirstChild("EspHighlight")
                if Configuration.EspEnabled then
                    if not highlight then
                        highlight = Instance.new("Highlight")
                        highlight.Name = "EspHighlight"
                        highlight.Parent = player.Character
                    end
                    highlight.FillColor = Configuration.EspColor
                    highlight.OutlineColor = Color3.fromRGB(255, 255, 255)
                    highlight.FillTransparency = 0.5
                    
                    local billboard = head:FindFirstChild("EspName")
                    if not billboard then
                        billboard = Instance.new("BillboardGui")
                        billboard.Name = "EspName"
                        billboard.Size = UDim2.new(0, 200, 0, 50)
                        billboard.StudsOffset = Vector3.new(0, 2, 0)
                        billboard.AlwaysOnTop = true
                        billboard.Parent = head
                        
                        local textLabel = Instance.new("TextLabel")
                        textLabel.Size = UDim2.new(1, 0, 1, 0)
                        textLabel.BackgroundTransparency = 1
                        textLabel.TextColor3 = Color3.fromRGB(255, 255, 255)
                        textLabel.TextStrokeTransparency = 0
                        textLabel.Font = Enum.Font.GothamBold
                        textLabel.TextSize = 14
                        textLabel.Parent = billboard
                    end
                    billboard.TextLabel.Text = player.DisplayName
                    billboard.TextLabel.TextColor3 = Configuration.EspColor
                else
                    if highlight then highlight:Destroy() end
                    if head:FindFirstChild("EspName") then head.EspName:Destroy() end
                end
            end
        end
    end
end

local CamlockToggleBtn = CreateActionBtn("Camlock: OFF", UDim2.new(0.075, 0, 0.2, 0), UDim2.new(0.55, 0, 0, 35), function()
    if isListening then return end
    Configuration.Locked = not Configuration.Locked
    if Configuration.Locked then
        targetPlayer = GetClosestPlayerToCursor()
        CamlockToggleBtn.Text = "Camlock: ON"
    else
        targetPlayer = nil
        CamlockToggleBtn.Text = "Camlock: OFF"
    end
end)

local CamlockKeyBtn
CamlockKeyBtn = CreateActionBtn("[" .. Configuration.CamlockKey.Name .. "]", UDim2.new(0.65, 0, 0.2, 0), UDim2.new(0.275, 0, 0, 35), function()
    if isListening then return end
    isListening = true
    CamlockKeyBtn.Text = "Press A Key"
    local conn
    conn = UserInputService.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.Keyboard then
            if input.KeyCode ~= Enum.KeyCode.Unknown then
                Configuration.CamlockKey = input.KeyCode
                CamlockKeyBtn.Text = "[" .. input.KeyCode.Name .. "]"
            else
                CamlockKeyBtn.Text = "[" .. Configuration.CamlockKey.Name .. "]"
            end
            isListening = false
            conn:Disconnect()
        end
    end)
end)

local ColorBtn = CreateActionBtn("Change ESP Color", UDim2.new(0.075, 0, 0.74, 0), UDim2.new(0.85, 0, 0, 35), function()
    if isListening then return end
    Configuration.EspColor = Color3.fromHSV(tick() % 5 / 5, 1, 1)
end)

UserInputService.InputBegan:Connect(function(input, processed)
    if isListening then return end
    if processed then return end
    if input.KeyCode == Configuration.CamlockKey then
        Configuration.Locked = not Configuration.Locked
        if Configuration.Locked then
            targetPlayer = GetClosestPlayerToCursor()
            CamlockToggleBtn.Text = "Camlock: ON"
        else
            targetPlayer = nil
            CamlockToggleBtn.Text = "Camlock: OFF"
        end
    elseif input.KeyCode == Configuration.GuiKey then
        MainFrame.Visible = not MainFrame.Visible
    elseif input.KeyCode == Configuration.EspKey then
        Configuration.EspEnabled = not Configuration.EspEnabled
        EspToggleBtn.Text = "ESP: " .. (Configuration.EspEnabled and "ON" or "OFF")
    end
end)

RunService.Heartbeat:Connect(function()
    if Configuration.Locked then
        if targetPlayer and targetPlayer.Character and targetPlayer.Character:FindFirstChild("Head") then
            local hum = targetPlayer.Character:FindFirstChildOfClass("Humanoid")
            if hum and hum.Health > 0 then
                local headPos = targetPlayer.Character.Head.Position
                Camera.CFrame = Camera.CFrame:Lerp(CFrame.new(Camera.CFrame.Position, headPos), Configuration.Smoothing)
            else
                targetPlayer = GetClosestPlayerToCursor()
            end
        else
            targetPlayer = GetClosestPlayerToCursor()
        end
    end
    UpdateEsp()
end)

Players.PlayerAdded:Connect(function(player)
    player.CharacterAdded:Connect(function()
        task.wait(0.5)
        UpdateEsp()
    end)
end)
